            <p>
              Match each function of <m>t</m> with its antiderivative. Assume <m>K</m> is a constant.
            </p>
          </statement>
          <cardsort>
            <match>
              <premise><me>f(t) = \frac{K}{t}</me></premise>
              <response><me>K\ln|t| + C</me></response>
            </match>
            <match>
              <premise><me>g(t) = \frac{1}{K}</me></premise>
              <response><me>\frac{t}{K} + C</me></response>
            </match>
            <match>
              <premise><me>h(t) = e^{Kt}</me></premise>
              <response><me>\frac{1}{K} e^{Kt} + C</me></response>
            </match>
            <match>
              <premise><me>h(t) = e^{K}</me></premise>
              <response><me>t e^{K} + C</me></response>
            </match>
            <match>
              <premise><me>m(t) = \frac{t}{K}</me></premise>
              <response><me>\frac{t^2}{2K} + C</me></response>
            </match>

              <response><me>\ln|K| + C</me></response>

      <subsubsection>
        <title>✔ Fill-in-the-Blank Questions</title>
        
        <exercise label="ex-demo-jscmp-primes">
        <title>Fill-In, Prime less then 20</title>
        <statement>
          <p>
            What is an example of a prime number less than 20?
            <fillin width="5" answer="13"/>
          </p>
        </statement>
        <evaluation>
          <evaluate>
            <test correct="yes">
              <jscmp><!-- test if number is in a list -->
                [2, 3, 5, 7, 11, 13, 17, 19].includes(Number(ans))
              </jscmp>
              <feedback>
                <p>
                  Any number from the list <m>\{2, 3, 5, 7, 11, 13, 17, 19\}</m> is a prime number less than 20.
                </p>
              </feedback>
            </test>
            <test>
              <jscmp><!-- find why it fails to be in list -->
                function(){
                  const val=Number(ans);
                  if (val &lt;= 0) { return "You need to give a positive integer." }
                  if (val &gt;= 20) { return "The integer must be less than 20." }
                  for (let i=2; i&lt;=3; i++) {
                  if (val % i == 0) { return `Your answer is composite since it is divisible by ${i}.` }
                }
                return false;
                }()
              </jscmp>
            </test>
          </evaluate>
        </evaluation>
        </exercise>
      </subsubsection>

          See how the code below counts the number of digits in an integer.

        <program label="partial-program-two" interactive="activecode" language="python">
          <code>
            num = 569
            nDigits = 0
            print(f"How many digits are in the integer {num}")
            temp = num
            while temp > 1:
              temp = temp/10
              nDigits = nDigits + 1
              print(f"  temp={temp}")

            print(f"{num} has {nDigits} digits")
